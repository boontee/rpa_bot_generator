defVar --name inputItems --type String
defVar --name operation --type String
defVar --name dosOutput --type String
defVar --name result1 --type String
defVar --name result2 --type String
defVar --name result3 --type String
defVar --name result4 --type String
defVar --name hasError --type Boolean

goSub --label Init
goSub --label CallPython
goSub --label ParseOutput
goSub --label PrintResults
goSub --label Cleanup

beginSub --name Init
  // Initialise variables
  setVar --name "${inputItems}" --value "Alice,Bob,Charlie,Diana"
  setVar --name "${operation}" --value "UPPER"
  setVar --name "${hasError}" --value "False"
  logMessage --message "Init: done" --type "Info"
endSub

beginSub --name CallPython
  // IBM RPA cannot call Python directly.
  // runDOSCommand invokes python.exe and captures all stdout as a single String — not a List.
  // Python must use print() to return values; WAL uses getRegex to extract them.
  logMessage --message "CallPython: invoking Python" --type "Info"
  runDOSCommand --command "python C:\\RPA\\scripts\\process_data.py ${inputItems} ${operation}"   dosOutput=value
  logMessage --message "CallPython: raw output=${dosOutput}" --type "Info"
  if --left "${dosOutput}" --operator "Equal_To" --right ""
    logMessage --message "CallPython: no output from Python" --type "Error"
    setVar --name "${hasError}" --value "True"
    goSub --label Cleanup
    throwError --message "Python script returned no output"
  endIf
  logMessage --message "CallPython: done" --type "Info"
endSub

beginSub --name ParseOutput
  // Python prints a single line: RESULT:ALICE,BOB,CHARLIE,DIANA
  // getRegex extracts each token by capture group position.
  // IBM RPA returns only a String from runDOSCommand, never a List,
  // so each result item is stored in its own defVar String variable.
  logMessage --message "ParseOutput: start" --type "Info"
  getRegex --text "${dosOutput}" --regex "RESULT:([^,\r\n]+)"   result1=value
  getRegex --text "${dosOutput}" --regex "RESULT:[^,\r\n]+,([^,\r\n]+)"   result2=value
  getRegex --text "${dosOutput}" --regex "RESULT:[^,\r\n]+,[^,\r\n]+,([^,\r\n]+)"   result3=value
  getRegex --text "${dosOutput}" --regex "RESULT:[^,\r\n]+,[^,\r\n]+,[^,\r\n]+,([^,\r\n]+)"   result4=value
  logMessage --message "ParseOutput: done" --type "Info"
endSub

beginSub --name PrintResults
  logMessage --message "PrintResults: start" --type "Info"
  logMessage --message "result1=${result1}" --type "Info"
  logMessage --message "result2=${result2}" --type "Info"
  logMessage --message "result3=${result3}" --type "Info"
  logMessage --message "result4=${result4}" --type "Info"
  logMessage --message "PrintResults: done" --type "Info"
endSub

beginSub --name ErrorHandler
  logMessage --message "ErrorHandler: error occurred" --type "Error"
  setVar --name "${hasError}" --value "True"
  goSub --label Cleanup
  throwError --message "An unexpected error occurred"
endSub

beginSub --name Cleanup
  if --left "${hasError}" --operator "Equal_To" --right "True"
    logMessage --message "Cleanup: finished with errors" --type "Warning"
  else
    logMessage --message "Cleanup: finished successfully" --type "Info"
  endIf
endSub
